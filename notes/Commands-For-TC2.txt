########################### Phase 6 #############################
# 1. Confirm the current number of nodes
kubectl get nodes
# 2. Create a temporary Deployment
kubectl create deployment capacity-test \
  --image=nginx:1.27 \
  --namespace tc2
# 3. Give each Pod a large CPU and memory request
kubectl set resources deployment capacity-test \
  --namespace tc2 \
  --requests=cpu=900m,memory=512Mi \
  --limits=cpu=900m,memory=512Mi
# 4. Scale the temporary Deployment to three Pods
kubectl scale deployment capacity-test \
  --namespace tc2 \
  --replicas=3
# 5. Check the Pods
kubectl get pods \
  --namespace tc2
# 6. Check for a second worker node periodically
kubectl get nodes
# 7. Recheck the Pods periodically
kubectl get pods \
  --namespace tc2
# 8. After the second node appears and all capacity-test Pods are Running,
# delete the temporary Deployment
kubectl delete deployment capacity-test \
  --namespace tc2
# 9. Confirm the temporary Pods were removed
kubectl get pods \
  --namespace tc2
# 10. Periodically check whether the extra node scales back down
kubectl get nodes
